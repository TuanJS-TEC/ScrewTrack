cmake_minimum_required(VERSION 3.16)
project(ScrewTrack CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs dnn video videoio)
find_package(SQLite3 REQUIRED)

file(GLOB SCREW_SRC CONFIGURE_DEPENDS src/*.cpp)
list(FILTER SCREW_SRC EXCLUDE REGEX "src/(main|eval_main|track_main|gui_main)\\.cpp$")
add_library(screw ${SCREW_SRC})
target_include_directories(screw PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(screw PUBLIC ${OpenCV_LIBS} SQLite::SQLite3)

if(EXISTS ${CMAKE_SOURCE_DIR}/src/main.cpp)
  add_executable(screwtrack src/main.cpp)
  target_link_libraries(screwtrack screw)
endif()
if(EXISTS ${CMAKE_SOURCE_DIR}/src/eval_main.cpp)
  add_executable(eval_labels src/eval_main.cpp)
  target_link_libraries(eval_labels screw)
endif()

if(EXISTS ${CMAKE_SOURCE_DIR}/src/track_main.cpp)
  add_executable(nuttrack src/track_main.cpp)
  target_link_libraries(nuttrack screw)
endif()

find_package(Qt6 COMPONENTS Widgets)
if(Qt6_FOUND AND EXISTS ${CMAKE_SOURCE_DIR}/src/gui_main.cpp)
  add_executable(nuttrack_gui src/gui_main.cpp)
  set_target_properties(nuttrack_gui PROPERTIES AUTOMOC ON)
  target_compile_definitions(nuttrack_gui PRIVATE PROJECT_DIR="${CMAKE_SOURCE_DIR}")
  target_link_libraries(nuttrack_gui screw Qt6::Widgets)
endif()

enable_testing()
file(GLOB TEST_SRC CONFIGURE_DEPENDS tests/test_*.cpp)
foreach(t ${TEST_SRC})
  get_filename_component(name ${t} NAME_WE)
  add_executable(${name} ${t})
  target_link_libraries(${name} screw)
  target_include_directories(${name} PRIVATE tests)
  target_compile_definitions(${name} PRIVATE PROJECT_DIR="${CMAKE_SOURCE_DIR}")
  add_test(NAME ${name} COMMAND ${name})
endforeach()
